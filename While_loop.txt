1. Print numbers from 1 to 5
i = 1
while i <= 5:
    print(i)
    i += 1


2. Print even numbers from 1 to 10
i = 1
while i <= 10:
    if i % 2 == 0:
        print(i)
    i += 1


3. Sum of first 5 numbers
i = 1
total = 0
while i <= 5:
    total += i
    i += 1
print("Sum:", total)


4. Reverse a number
num = 123
rev = 0
while num > 0:
    digit = num % 10
    rev = rev * 10 + digit
    num //= 10
print("Reversed:", rev)


5. Palindrome check
num = 121
temp = num
rev = 0
while num > 0:
    digit = num % 10
    rev = rev * 10 + digit
    num //= 10
if temp == rev:
    print("Palindrome")
else:
    print("Not Palindrome")
